Linear Search & Binary Search

C Language
#include<stdio.h>

int linearSearch(int arr[], int n, int key){
    for(int i=0; i<n; i++)
        if(arr[i]==key)
            return i;
    return -1;
}

int binarySearch(int arr[], int n, int key){
    int low=0, high=n-1, mid;
    while(low<=high){
        mid=(low+high)/2;
        if(arr[mid]==key)
            return mid;
        else if(arr[mid]<key) low=mid+1;
        else high=mid-1;
    }
    return -1;
}

Bubble Sort & Selection Sort

C Language
#include<stdio.h>

void bubbleSort(float arr[], int n){
    int i, j;
    float temp;
    int swapped;
    for(i = 0; i < n-1; i++){
        swapped = 0;
        for(j = 0; j < n-i-1; j++){
            if(arr[j] > arr[j+1]){
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
                swapped = 1;
            }
        }
        if(swapped == 0)
            break;
    }
}

void selectionSort(float arr[], int n){
    int i, j, minIndex;
    float temp;
    for(i = 0; i < n-1; i++){
        minIndex = i;
        for(j = i+1; j < n; j++){
            if(arr[j] < arr[minIndex])
                minIndex = j;
        }
        temp = arr[i];
        arr[i] = arr[minIndex];
        arr[minIndex] = temp;
    }
}

void display(float arr[], int n){
    for(int i = 0; i < n; i++)
        printf("%.2f ", arr[i]);
    printf("\\n");
}

void topFive(float arr[], int n){
    printf("Top 5 Scores: ");
    int count = (n < 5)? n : 5;
    for(int i = n-1; i >= n-count; i--)
        printf("%.2f ", arr[i]);
    printf("\\n");
}

Hashing Techniques

#include <stdio.h>

int hashtable[10];

void linear_insert(int key){
    int index;
    index = key % 10;
    while(hashtable[index]!= -1){
        index = (index + 1) % 10;
    }
    hashtable[index] = key;
}

void quadratic_insert(int key){
    int index;
    int i = 0;
    while(i <= 10){
        index = key % 10;
        index = (index + (i * i)) % 10;
        if(hashtable[index] == -1){
            hashtable[index] = key;
            break;
        } else {
            i++;
        }
    }
}

void double_insert(int key){
    int index;
    int i = 0;
    int h1 = key % 10;
    int h2 = 7 - (key % 7);
    while(i <= 10){
        index = key % 10;
        index = (h1 + (i * h2)) % 10;
        if(hashtable[index] == -1){
            hashtable[index] = key;
            break;
        } else {
            i++;
        }
    }
}

void display(){

}

Cinema Ticket Booking – Doubly Linked List

C Language
#include <stdio.h>
#include <stdlib.h>

struct node {
    int seat_no;
    struct node *prev;
    struct node *next;
};

struct node *head[10] = {NULL};

void create() {
    for (int row = 0; row < 10; row++) {
        if (head[row] != NULL) continue;
        struct node *temp = NULL;
        for (int i = 1; i <= 7; i++) {
            struct node *newnode = (struct node*)malloc(sizeof(struct node));
            newnode->seat_no = i;
            newnode->prev = NULL;
            newnode->next = NULL;
            if (head[row] == NULL) {
                head[row] = newnode;
                temp = newnode;
            } else {
                temp->next = newnode;
                newnode->prev = temp;
                temp = newnode;
            }
        }
    }
}

void book() {
    int row, s;
    printf("\\nEnter row (1-10): ");
    scanf("%d", &row);
    printf("Enter seat_no to be booked (1-7): ");
    scanf("%d", &s);
    if (row < 1 || row > 10) {
        printf("Invalid row!\\n");
        return;
    }
    struct node *temp = head[row-1];
    while (temp != NULL && temp->seat_no != s) {
        temp = temp->next;
    }
    if (temp == NULL) {
        printf("Seat %d in row %d is already booked!\\n", s, row);
        return;
    }
    if (temp == head[row-1]) {
        head[row-1] = temp->next;
        if (head[row-1] != NULL) {
            head[row-1]->prev = NULL;
        }
    } else if (temp->next == NULL) {
        temp->prev->next = NULL;
    } else {
        temp->prev->next = temp->next;
        temp->next->prev = temp->prev;
    }
    free(temp);
    printf("Seat %d in row %d booked successfully!\\n", s, row);
}

void cancel() {
    int row, s;
    printf("\\nEnter row (1-10) to cancel: ");
    scanf("%d", &row);
    printf("Enter seat_no to cancel (1-7): ");
    scanf("%d", &s);
    if (row < 1 || row > 10 || s < 1 || s > 7) {
        printf("Invalid input!\\n");
        return;

    struct node *check = head[row-1];
    while (check != NULL) {
        if (check->seat_no == s) {
            printf("Seat already available!\\n");
            return;
        }
        check = check->next;
    }
    struct node *newnode = (struct node*)malloc(sizeof(struct node));
    newnode->seat_no = s;
    newnode->prev = NULL;
    newnode->next = NULL;

    if (head[row-1] == NULL || s < head[row-1]->seat_no) {
        newnode->next = head[row-1];
        if (head[row-1] != NULL) {
            head[row-1]->prev = newnode;
        }
        head[row-1] = newnode;
    } else {
        struct node *temp = head[row-1];
        while (temp->next != NULL && temp->next->seat_no < s) {
            temp = temp->next;
        }
        newnode->next = temp->next;
        newnode->prev = temp;
        if (temp->next != NULL) {
            temp->next->prev = newnode;
        }
        temp->next = newnode;
    }
    printf("Booking for seat %d in row %d cancelled!\\n", s, row);
}

void display() {

Infix to Postfix – Stack

C Language
#include <stdio.h>
#include <ctype.h>
#define MAX 50

char stack[MAX];
int top = -1;

int isfull() {
    return top == MAX - 1;
}

int isempty() {
    return top == -1;
}

void push(char x) {
    if (isfull()) {
        printf("Stack is full\\n");
    } else {
        stack[++top] = x;
    }
}

char pop() {
    if (isempty()) {
        printf("Stack is empty\\n");
        return '\\0';
    } else {
        return stack[top--];
    }
}

char peek() {
    if (isempty()) return '\\0';
    return stack[top];
}

int priority(char ch) {
    if (ch == '+' || ch == '-')
        return 1;
    if (ch == '*' || ch == '/' || ch == '%')
        return 2;
    if (ch == '^')
        return 3;
    return 0;
}

void convert(char infix[], char postfix[]) {
    int i = 0, j = 0;
    char ch;
    while ((ch = infix[i++]) != '\\0') {
        if (isalnum(ch)) {
            postfix[j++] = ch;
        }
        else if (ch == '(') {
            push(ch);
        }
        else if (ch == ')') {
            while (!isempty() && peek() != '(') {
                postfix[j++] = pop();
            }
            if (!isempty() && peek() == '(') {
                pop();
            }
        }
        else {
            while (!isempty() && priority(peek()) >= priority(ch) && peek() != '(') {
                if (ch == '^' && priority(peek()) == priority(ch)) break;
                postfix[j++] = pop();
            }
            push(ch);
        }
    }
    while (!isempty()) {
        postfix[j++] = pop();
    }
    postfix[j] = '\\0';
}

int main() {
    char infix[MAX], postfix[MAX];
    printf("Enter infix expression: ");
    scanf("%s", infix);
    convert(infix, postfix);
    printf("Infix: %s\\n", infix);
    printf("Postfix: %s\\n", postfix);
    return 0;
}

Pizza Parlour – Linear Queue

#include <stdio.h>
#define SIZE 10

    if (rear == SIZE - 1)
        return 1;
    else
        return 0;

    if (front == -1)
        return 1;
    else
        return 0;
}

void order() {
    int pizza;
    if (isFull()) {
        printf("Pizza parlour is full! Cannot take more orders.\\n");
        return;
    }
    printf("Enter the pizza order number: ");
    scanf("%d", &pizza);
    if (isEmpty()) {
        front = 0;
    }
    rear++;
    queue[rear] = pizza;
    printf("Order %d placed successfully!\\n", pizza);
}

void serveOrder() {
    if (isEmpty()) {
        printf("Pizza parlour is empty! No orders to serve.\\n");
        return;
    }
    printf("Order %d is served.\\n", queue[front]);
    if (front == rear) {
        front = rear = -1;
    } else {
        front++;
    }
}

void display() {
    if (isEmpty()) {
        printf("No orders available.\\n");
        return;
    }
    printf("Orders: Front -> ");
    for (int i = front; i <= rear; i++) {
        printf("%d ", queue[i]);
    }
    printf("<- Rear\\n");
}

    }
}

Binary Search Tree (BST)

C Language
#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *left;
    struct node *right;
};

struct node *createnode(int value) {
    struct node *newnode = (struct node*)malloc(sizeof(struct node));
    newnode->data = value;
    newnode->left = NULL;
    newnode->right = NULL;
    return newnode;
}

struct node *insert(struct node *root, int value) {
    if (root == NULL) {
        return createnode(value);
    }
    if (value < root->data) {
        root->left = insert(root->left, value);
    } else if (value > root->data) {
        root->right = insert(root->right, value);
    }
    return root;
}

void inorder(struct node *root) {
    if (root != NULL) {
        inorder(root->left);
        printf("%d  ", root->data);
        inorder(root->right);
    }
}

int height(struct node *root) {
    if (root == NULL) return 0;
    int hleft = height(root->left);
    int hright = height(root->right);
    return (hleft > hright ? hleft : hright) + 1;
}

int min(struct node *root) {
    if (root == NULL) {
        printf("Tree is empty\\n");
        return -1;
    }
    struct node *temp = root;
    while (temp->left != NULL) {
        temp = temp->left;
    }
    return temp->data;
}

int search(struct node *root, int key) {
    if (root == NULL) {
        return 0;
    }
    if (key == root->data) {
        return 1;
    }
    if (key < root->data) {
        return search(root->left, key);
    } else {
        return search(root->right, key);
    }
}

void mirror(struct node *root) {
    if (root != NULL) {
        struct node *temp = root->left;
        root->left = root->right;
        root->right = temp;
        mirror(root->left);
        mirror(root->right);
    }
}

int main() {
    struct node *root = NULL;
    root = insert(root, 50);
    insert(root, 30);
    insert(root, 60);
    insert(root, 10);
    insert(root, 35);
    insert(root, 70);
    insert(root, 15);
    insert(root, 85);
    insert(root, 40);

    printf("Inorder sequence of Tree is :\\n");
    inorder(root);

    int ch, key, value;
    do {
        printf("\\n\\n1. Insert a new node\\n2. Find height\\n3. Find minimum\\n4. Mirror a tree\\n5. Search a value\\n6. EXIT \\nEnter choice : ");
        scanf("%d", &ch);
        switch (ch) {
            case 1:
                printf("Enter value to insert: ");
                scanf("%d", &value);
                root = insert(root, value);
                inorder(root);
                break;
            case 2:
                printf("\\nHeight of tree is : %d", height(root));
                break;
            case 3:
                printf("\\nMinimum data in tree is : %d", min(root));
                break;
            case 4:
                printf("Mirror of tree (Inorder):\\n");
                mirror(root);
                inorder(root);
                mirror(root);
                break;
            case 5:
                printf("Enter the value of key : ");
                scanf("%d", &key);
                if (search(root, key))
                    printf("Element %d found!\\n", key);
                else
                    printf("Element %d NOT found!\\n", key);
                break;
            case 6:
                exit(0);
            default:
                printf("Invalid choice\\n");
        }
    } while (ch != 6);

}

void createGraph() {
    printf("\\nEnter number of cities (max %d): ", MAX);
    scanf("%d", &n);

    for(int i=0; i<n; i++) {
        printf("Enter name of city %d: ", i+1);
        scanf("%s", cityName[i]);
    }

    for(int i=0; i<n; i++) {
        for(int j=0; j<n; j++) {
            adj[i][j] = 0;
        }
    }

    int edges;
    printf("\\nEnter number of flight paths: ");
    scanf("%d", &edges);
    printf("Enter flight paths (Source Destination Cost):\\n");
    printf("Use city numbers 0 to %d\\n", n-1);
    for(int k=0; k<edges; k++) {
        int src, dest, cost;
        printf("Flight %d: ", k+1);
        scanf("%d %d %d", &src, &dest, &cost);
        if(src >=0 && src < n && dest >=0 && dest < n) {
            adj[src][dest] = cost;
            adj[dest][src] = cost;
        } else {
            printf("Invalid city number!\\n");
            k--;
        }
    }
    printf("\\nGraph Created Successfully!\\n");
}

void addEdge() {
    int src, dest, cost;
    printf("\\nEnter new flight (Source Destination Cost/Time/Fuel): ");
    scanf("%d %d %d", &src, &dest, &cost);
    if(src >=0 && src < n && dest >=0 && dest < n) {
        adj[src][dest] = cost;
        adj[dest][src] = cost;
        printf("Flight added between %s and %s with cost %d\\n", cityName[src], cityName[dest], cost);
    } else {
        printf("Invalid cities!\\n");
    }
}

void displayGraph() {
    printf("\\n--- Adjacency Matrix (Cost = Time/Fuel) ---\\n");
    printf("\\t");
    for(int i=0; i<n; i++) printf("%s\\t", cityName[i]);
    printf("\\n");

    for(int i=0; i<n; i++) {
        printf("%s\\t", cityName[i]);
        for(int j=0; j<n; j++) {
            printf("%d\\t", adj[i][j]);
        }
        printf("\\n");
    }

    printf("\\n--- Adjacency List ---\\n");
    for(int i=0; i<n; i++) {
        printf("%s -> ", cityName[i]);
        for(int j=0; j<n; j++) {
            if(adj[i][j] != 0) {
                printf("%s(%d) -> ", cityName[j], adj[i][j]);
            }
        }
        printf("NULL\\n");
    }
}

void DFS(int v) {
    visited[v] = 1;
    for(int i=0; i<n; i++) {
        if(adj[v][i] != 0 && !visited[i]) {
            DFS(i);
        }
    }
}

void resetVisited() {
    for(int i=0; i<n; i++) visited[i] = 0;
}

int isConnected() {
    resetVisited();
    DFS(0);
    for(int i=0; i<n; i++) {
        if(!visited[i]) {
            return 0;
        }
    }
    return 1;
}

int main() {
    int ch;
    createGraph();

    while(1) {
        printf("\\n\\n~~~~ FLIGHT GRAPH MENU ~~~~\\n");
        printf("1. Display Graph (Matrix & List)\\n");
        printf("2. Add New Flight Path\\n");
        printf("3. Check if Graph is Connected\\n");
        printf("4. Exit\\n");
        printf("Enter your choice: ");
        scanf("%d", &ch);

        switch(ch) {
            case 1:
                displayGraph();
                break;
            case 2:
                addEdge();
                break;
            case 3:
                if(isConnected()) {
                    printf("\\n*** Graph is CONNECTED - You can travel between all cities ***\\n");
                } else {
                    printf("\\n*** Graph is NOT CONNECTED - Some cities are isolated ***\\n");
                    printf("Unreachable cities from %s: ", cityName[0]);
                    for(int i=0; i<n; i++) {
                        if(!visited[i]) printf("%s ", cityName[i]);
                    }
                    printf("\\n");
                }
                break;
            case 4:
                printf("Exiting...\\n");
                exit(0);
            default:
                printf("Invalid choice!\\n");
        }
    }
    return 0;
}
